# The vital_signs_analyzer script implements the following requirements:
# 1/ The system should accept heart rate (HR) and blood oxygen saturation (SpO2) readings. The values should be normalized into a 0-1 range using defined physiological limits.
# 2/ The system should detect the patient's status based on thresholds for normalized HR and SpO2.
# 3/ The user should be able to override the status with a manually set value.
# 4/ The system should support displaying the status (if enabled) and logging it to a file (if enabled).

last_values = {}
HR_MIN = 40
HR_MAX = 180
SPO2_MIN = 0
SPO2_MAX = 100
DEBUG_MODE = False


def normalize_hr(hr):
    """
    Normalizes heart rate to a 0–1 scale based on physiological limits.

    Parameters:
        hr (int/float): Heart rate in beats per minute.

    Returns:
        float: Normalized value between 0 and 1.

    Raises:
        ValueError: If HR is outside reasonable bounds.
    """
    if hr < HR_MIN or hr > HR_MAX:
        raise ValueError(f"Heart rate {hr} is outside normal range ({HR_MIN}-{HR_MAX})")
    return (hr - HR_MIN) / (HR_MAX - HR_MIN)


def normalize_spo2(spo2):
    """
    Normalizes SpO2 to a 0–1 scale (percentage to decimal).

    Parameters:
        spo2 (int/float): Oxygen saturation percentage (0–100).

    Returns:
        float: Normalized value between 0 and 1.

    Raises:
        ValueError: If SpO2 is outside 0–100.
    """
    if spo2 < 0 or spo2 > 100:
        raise ValueError(f"SpO2 {spo2} is outside valid range (0-100%)")
    return spo2 / 100.0


def analyze_vitals(hr, spo2, log=True, override_status=None, verbose=False):
    """
    Analyzes heart rate and SpO2 to determine patient status.

    Parameters:
        hr (int/float): Heart rate.
        spo2 (int/float): Oxygen saturation percentage.
        log (bool): Whether to log results to a file.
        override_status (str): Manually set status (optional).
        verbose (bool): Whether to print detailed messages.

    Returns:
        str: Status message.

    Raises:
        ValueError: If inputs are invalid.
    """
    global last_values

    # Input validation
    if hr <= 0 or spo2 < 0:
        raise ValueError("HR must be positive and SpO2 cannot be negative")

    last_values["hr"] = hr
    last_values["spo2"] = spo2

    if verbose:
        print(f"Analyzing vital data: HR={hr}, SpO2={spo2}")

    # Normalize
    norm_hr = normalize_hr(hr)
    norm_spo2 = normalize_spo2(spo2)

    # Determine status
    if override_status:
        status = override_status
    else:
        if norm_hr > 0.7 and norm_spo2 < 0.3:
            status = "CRITICAL"
        elif norm_hr > 0.4:
            if spo2 < 88 or hr > HR_MAX:
                status = "SEVERE"
            else:
                status = "WARNING"
        else:
            status = "OK"

    # Update global last status string
    global tmp
    tmp = f"Last status: {status}" + (" (DEBUG)" if DEBUG_MODE else "")

    # Log to file if requested
    if log:
        try:
            with open("vitals_log.txt", "a") as f:
                f.write(f"HR={hr}, SpO2={spo2}, N_HR={norm_hr:.2f}, N_SpO2={norm_spo2:.2f}, STATUS={status}\n")
        except IOError as e:
            print(f"Failed to write log: {e}")

    # Display status if verbose
    if verbose:
        print(f"Status is: {status}")

    return tmp


if __name__ == "__main__":
    # Example with valid data
    hr_example = 85
    spo2_example = 97

    result = analyze_vitals(hr_example, spo2_example, log=True, verbose=True)
    print("Analysis result:", result)
    print("Last values:", last_values)
